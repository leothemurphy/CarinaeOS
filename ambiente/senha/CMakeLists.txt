# carinae-senha — o agente de autenticação (polkit) do CarinaeOS: a janela "Autorizar"
cmake_minimum_required(VERSION 3.16)
project(carinae-senha VERSION 0.1 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POLKIT REQUIRED IMPORTED_TARGET polkit-agent-1 polkit-gobject-1)

qt_add_executable(carinae-senha main.cpp)
qt_add_resources(carinae-senha "qml" PREFIX "/senha" FILES Janela.qml)
target_link_libraries(carinae-senha PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick PkgConfig::POLKIT)
install(TARGETS carinae-senha RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
