# carinae-bloqueio — o ecrã de bloqueio do CarinaeOS (relógio, avatar, campo de senha; PAM)
cmake_minimum_required(VERSION 3.16)
project(carinae-bloqueio VERSION 0.1 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCBB REQUIRED IMPORTED_TARGET xcb)
find_library(PAM_LIB pam REQUIRED)
find_path(PAM_INC security/pam_appl.h REQUIRED)

qt_add_executable(carinae-bloqueio main.cpp)
qt_add_resources(carinae-bloqueio "qml" PREFIX "/bloqueio" FILES Ecra.qml)
target_include_directories(carinae-bloqueio PRIVATE ${PAM_INC})
target_link_libraries(carinae-bloqueio PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick PkgConfig::XCBB ${PAM_LIB})
install(TARGETS carinae-bloqueio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# o serviço PAM tem de estar onde o PAM o procura (fora do prefixo)
install(FILES carinae-bloqueio.pam DESTINATION /etc/pam.d RENAME carinae-bloqueio)
