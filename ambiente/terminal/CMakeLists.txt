# carinae-terminal — o Terminal Carinae (motor libvterm + desenho próprio; sem o Konsole)
cmake_minimum_required(VERSION 3.16)
project(carinae-terminal VERSION 0.1 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VTERM REQUIRED IMPORTED_TARGET vterm)

qt_add_executable(carinae-terminal main.cpp terminal.cpp terminal.h)
qt_add_resources(carinae-terminal "qml" PREFIX "/terminal" FILES Main.qml)
target_link_libraries(carinae-terminal PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 PkgConfig::VTERM util)
install(TARGETS carinae-terminal RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES carinaeos-terminal.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
